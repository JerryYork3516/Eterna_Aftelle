def evaluate(lo, hi):
    expected = hi - lo
    assert expected > 0 and hi <= min(len(u), len(delta_d), len(delta_a))
    valid = (v['mixed'][lo:hi] != 0) & (v['background'][lo:hi] != 0)
    support_complete = len(valid) == expected and bool(valid.all())
    energy = float(np.sum(u[lo:hi] ** 2))
    result = {'source16': [lo, hi], 'samples': expected, 'source_energy': energy,
        'full_DTLN_mask_support': support_complete, 'valid_samples': int(valid.sum()),
        'AEC_domain_in_bounds': True, 'unscaled_source_RMS': float(np.sqrt(energy / expected))}
    if not support_complete:
        result['status'] = 'UNAVAILABLE_SUPPORT_STOP'
        return result
    if energy == 0:
        result['status'] = 'NO_ACTIVE_SOURCE'
        return result
    for name, response in [('DTLN', delta_d), ('AEC3', delta_a)]:
        z = response[lo:hi]
        response_energy = float(np.sum(z ** 2))
        result[name] = {'E': float(np.sum((z - u[lo:hi]) ** 2) / energy),
            'g': float(np.dot(z, u[lo:hi]) / energy),
            'response_RMS': float(np.sqrt(response_energy / expected)),
            'correlation': None if response_energy == 0 else float(np.dot(z, u[lo:hi]) / np.sqrt(response_energy * energy))}
    result['DTLN_E_not_above_AEC3'] = result['DTLN']['E'] <= result['AEC3']['E']
    result['status'] = 'SOURCE_RESPONSE_NO_WORSE' if result['DTLN_E_not_above_AEC3'] else 'SOURCE_RESPONSE_DEGRADED_STOP'
    return result


def ledger_for_frames(frame_ids):
    report = {}
    for role, m in metrics.items():
        frame_ids = set(frame_ids)
        chains = [{'original_full_chain': c['sourceFrames'],
                   'frames_in_interval': sorted(frame_ids & set(c['sourceFrames']))}
                  for c in m['confirmation_chains'] if frame_ids & set(c['sourceFrames'])]
        states = [traces[role][f] for f in sorted(frame_ids)]
        branches = {}
        for state in states:
            name = state['actualBranch']
            branches[name] = branches.get(name, 0) + 1
        report[role] = {'qualified': sorted(frame_ids & set(m['qualified_source_ids'])),
            'Gate': sorted(frame_ids & set(m['Gate_source_ids'])),
            'forwarded_by_own_sourceID': sorted(frame_ids & set(m['forwarded_source_ids'])),
            'confirm_chain_intersections': chains, 'branches': branches,
            'confirmationCount_after_by_sourceID': {str(s['sourceFrame']): s['afterGate']['confirmationCount'] for s in states},
            'origin': 'Frozen prior Host replay, not rerun or new qualifications'}
    return report

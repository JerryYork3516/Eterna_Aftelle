def save(name, data):
    path = WORK / name
    assert not path.exists(), 'Do not overwrite frozen evidence'
    path.write_text(json.dumps(data, indent=2, allow_nan=False) + '\n')


def exact(a, b):
    return np.array_equal(np.asarray(a, dtype='<f4').view('<u4'), np.asarray(b, dtype='<f4').view('<u4'))

raw, cache, full_separate = {}, {}, {}
for arm, capsule in CAPS.items():
    raw[arm] = {kind: np.fromfile(capsule / name, dtype='<f4') for kind, name in

    reached = False

        assert call['sampleOffset'] == available[kind]

        safe_end = max(0, (available[kind] - 31) // 3 + 1)
        now = resample_poly(raw[arm][kind][:available[kind]], 1, 3).astype('<f4')[:safe_end]
        low[kind] = np.concatenate((low[kind], now[len(low[kind]):]))

def parts(z, target):
    z, target = np.asarray(z, dtype=float), np.asarray(target, dtype=float)
    den = float(np.sum(target ** 2))
    if den == 0:
        return {'status': 'NO_ACTIVE_SOURCE_VIEW', 'samples': len(target), 'source_energy': 0}
    energy = float(np.sum(z ** 2))
    return {'status': 'VALID', 'samples': len(target), 'source_energy': den,
            'g_signed_projection': float(np.dot(z, target) / den), 'energy': energy,
            'energy_to_source_ratio': energy / den,
            'RMS': float(np.sqrt(energy / len(z))), 'max_abs': float(np.max(np.abs(z)))}


def response(z, target):
    r = parts(z, target)
    if r['status'] == 'VALID':
        r['E'] = float(np.sum((np.asarray(z, dtype=float) - target) ** 2) / r['source_energy'])
    return r


views = []
for i in RECORDS:
    b = bindings['mixed'][i]
    block_lo, block_hi = b['mic512_actual_real_source16']
    lo, hi = max(J[0], block_lo), min(J[1], block_hi)
    local = slice(lo - block_lo, hi - block_lo)
    target = u_full[lo:hi]
    mic_delta = arrays['mixed']['mic'][i][local].astype(float) - arrays['background']['mic'][i][local].astype(float)
    expected_input = cached_pair[lo:hi]
    assert np.array_equal(mic_delta, expected_input)
    cache_difference = cached_pair[lo:hi] - separate_pair[lo:hi]
    order_difference = separate_pair[lo:hi] - target
    input_residual = mic_delta - target - (cache_difference + order_difference)
    assert np.max(np.abs(input_residual)) <= 1e-12
    stage_delta = arrays['mixed']['stage1'][i][local].astype(float) - arrays['background']['stage1'][i][local].astype(float)
    xm = np.fft.rfft(arrays['mixed']['mic'][i]).astype('complex64')
    xb = np.fft.rfft(arrays['background']['mic'][i]).astype('complex64')
    wm, wb = arrays['mixed']['mask'][i], arrays['background']['mask'][i]
    input_term_spectrum = (xm - xb) * wb
    mask_term_spectrum = xm * (wm - wb)
    actual_spectrum_difference = (xm * wm) - (xb * wb)
    spectral_rounding = actual_spectrum_difference.astype('complex128') - (input_term_spectrum.astype('complex128') + mask_term_spectrum.astype('complex128'))
    input_term = np.fft.irfft(input_term_spectrum).astype('<f4')[local].astype(float)
    mask_term = np.fft.irfft(mask_term_spectrum).astype('<f4')[local].astype(float)
    stage_rounding = stage_delta - (input_term + mask_term)
    assert np.max(np.abs(stage_delta - (input_term + mask_term + stage_rounding))) <= 1e-12
    rm, rs = response(mic_delta, target), response(stage_delta, target)
    observed = None if rm['status'] != 'VALID' else (
        rs['g_signed_projection'] < rm['g_signed_projection'] and rs['E'] > rm['E'])
    views.append({'record': i, 'source16': [lo, hi], 'local512_slice': [local.start, local.stop],
        'J_slice': [lo - J[0], hi - J[0]], 'input_ready_call': 141,
        'mic_pair': rm, 'Stage1_pair': rs, 'Stage1_added_decrease_witness': observed,
        'input_precision': {'cache_vs_separate_full': parts(cache_difference, target),
            'separate_full_vs_original_u': parts(order_difference, target),
            'closure_max_abs': float(np.max(np.abs(input_residual))),
            'u_changed': False, 'actual_cache_bit_exact': True},
        'mask_algebra': {'actual_contexts_and_masks_only': True,
            'input_change_under_saved_background_mask': parts(input_term, target),
            'actual_mask_change_term': parts(mask_term, target),
            'Float32_spectral_rounding_max_abs': float(np.max(np.abs(spectral_rounding))),
            'time_domain_rounding_residual': parts(stage_rounding, target),
            'not_a_mask_substitution_model_experiment': True}})

contributions, details = [], []

            index = r - 3 + rank

def bit_exact(a, b):
    return np.array_equal(np.asarray(a, dtype='<f4').view('<u4'),
                          np.asarray(b, dtype='<f4').view('<u4'))

def tensor_contract(details):
    return [{k: (v.tolist() if isinstance(v, np.ndarray) else str(v) if k == 'dtype' else v)
             for k, v in d.items() if k in ['name', 'index', 'shape', 'shape_signature', 'dtype']}
            for d in details]


def metrics(pair):
    den = float(np.dot(u, u))
    return {'samples': len(pair), 'source_energy': den,
        'E': float(np.dot(pair - u, pair - u) / den),
        'g': float(np.dot(pair, u) / den)}


class CheckedInterpreter:
    def __init__(self, role, resolver):
        self.role, self.record = role, None
        try:
            self.i = tflite.Interpreter(model_path=str(OFFICIAL / f'dtln_aec_512_{role}.tflite'),
                num_threads=1, experimental_op_resolver_type=resolver)
            self.i.allocate_tensors()
        except Exception as exc:
            raise RuntimeError(f'Stage{role} initialize/allocate: {type(exc).__name__}: {exc}') from exc
        self.ins, self.outs = self.i.get_input_details(), self.i.get_output_details()
        assert tensor_contract(self.ins) == metadata[str(role)]['inputs']
        assert tensor_contract(self.outs) == metadata[str(role)]['outputs']
        self.prior = None

    def set_tensor(self, index, value):
        assert np.isfinite(value).all()
        if index == self.ins[1]['index']:
            assert not np.any(value) if self.prior is None else bit_exact(value, self.prior)
        self.i.set_tensor(index, value)

    def invoke(self):
        try:
            self.i.invoke()
        except Exception as exc:
            raise RuntimeError(f'Stage{self.role} record{self.record} invoke: {type(exc).__name__}: {exc}') from exc

    def get_tensor(self, index):
        value = self.i.get_tensor(index)
        assert np.isfinite(value).all()
        if index == self.outs[1]['index']:
            self.prior = value.copy()
        return value


def namespace(resolver):
    a, b = CheckedInterpreter(1, resolver), CheckedInterpreter(2, resolver)
    contracts = {str(role): {'inputs': tensor_contract(i.ins), 'outputs': tensor_contract(i.outs),
        'ops': [d['op_name'] for d in i.i._get_ops_details()]}
        for role, i in [(1, a), (2, b)]}
    assert not any('DELEGATE' in v['ops'] for v in contracts.values())
    return {'np': np, 'interpreter_1': a, 'interpreter_2': b,
        'input_details_1': a.ins, 'output_details_1': a.outs,
        'input_details_2': b.ins, 'output_details_2': b.outs,
        'states_1': np.zeros(a.ins[1]['shape'], dtype=np.float32),
        'states_2': np.zeros(b.ins[1]['shape'], dtype=np.float32),
        'in_buffer': np.zeros(512, dtype=np.float32), 'in_buffer_lpb': np.zeros(512, dtype=np.float32),
        'out_buffer': np.zeros(512, dtype=np.float32), 'out_file': np.zeros(128),
        'block_len': 512, 'block_shift': 128, 'idx': 0}, contracts


def run_arm(arm, resolver, original, optimized=None):
    ns, contracts = namespace(resolver)
    rows, differences = [], []
    for r in range(337):
        ns['interpreter_1'].record = ns['interpreter_2'].record = r
        incoming = [ns[k].copy() for k in ['states_1', 'states_2']]
        ns['audio'], ns['lpb'] = arrays[arm]['mic'][r][-128:].copy(), arrays[arm]['ref'][r][-128:].copy()
        exec(core, ns)
        values = {role: np.asarray(ns[name], dtype='<f4').reshape(-1).copy() for role, name in
            [('mic', 'in_buffer'), ('ref', 'in_buffer_lpb'), ('mask', 'out_mask'),
             ('stage1', 'estimated_block'), ('stage2', 'out_block'), ('final', 'out_file')]}
        values.update({f'state_in_{k}': incoming[k - 1] for k in [1, 2]})
        values.update({f'state_out_{k}': ns[f'states_{k}'].copy() for k in [1, 2]})
        assert all(np.isfinite(v).all() for v in values.values())
        for role in ['mic', 'ref']:
            assert bit_exact(values[role], arrays[arm][role][r]), ('Rebuilt input', arm, r, role)
        for role, name in [('mic', 'in_block_fft'), ('ref', 'lpb_block_fft')]:
            assert digest(ns[name]) == old_states[arm][r][f'{role}_fft_sha256']
        if original:
            for role in ROLES:
                assert bit_exact(values[role], arrays[arm][role][r]), ('Original output', arm, r, role)
            for direction in ['in', 'out']:
                for k in [1, 2]:
                    assert digest(values[f'state_{direction}_{k}']) == old_states[arm][r][f'state_{direction}'][k - 1], ('Original state', arm, r, direction, k)
        else:
            for role in ['mask', 'state_out_1', 'stage1', 'stage2', 'state_out_2', 'final']:
                a, b = values[role], optimized[r][role]
                if not bit_exact(a, b):
                    indices = np.flatnonzero(a.reshape(-1).view('<u4') != b.reshape(-1).view('<u4'))
                    delta = a.astype(float) - b.astype(float)
                    differences.append({'record': r, 'role': role, 'shape': list(a.shape),
                        'different_elements': int(len(indices)), 'first_flat_index': int(indices[0]),
                        'maximum_absolute_difference': float(np.max(np.abs(delta))),
                        'reference_sha256': digest(a), 'optimized_sha256': digest(b)})
        rows.append(values)
        if r in [0, 112, 224, 336]:
            print(json.dumps({'endpoint': resolver.name, 'arm': arm, 'records_completed': r + 1}), flush=True)
    return rows, contracts, differences

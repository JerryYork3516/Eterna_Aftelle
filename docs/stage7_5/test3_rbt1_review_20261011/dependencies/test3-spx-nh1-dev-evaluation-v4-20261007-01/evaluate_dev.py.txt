def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False,
                      allow_nan=False).encode('utf-8')


def load(path):
    return json.loads(Path(path).read_text())


def publish_json(path, value):
    pending = Path(path).with_name(Path(path).name + '.pending')
    with pending.open('x') as handle:
        json.dump(value, handle, indent=2, ensure_ascii=False, allow_nan=False)
        handle.write('\n')
        handle.flush()
        os.fsync(handle.fileno())
    os.link(pending, path)


def publish_array(np, path, array):
    pending = Path(path).with_name(Path(path).name + '.pending')
    with pending.open('xb') as handle:
        np.save(handle, array, allow_pickle=False)
        handle.flush()
        os.fsync(handle.fileno())
    os.link(pending, path)


def publish_npz(np, path, **arrays):
    pending = Path(path).with_name(Path(path).name + '.pending')
    with pending.open('xb') as handle:
        np.savez_compressed(handle, **arrays)
        handle.flush()
        os.fsync(handle.fileno())
    os.link(pending, path)

def f32(value):
    return struct.unpack('<f', struct.pack('<f', value))[0]


def overlap(intervals, start, end):
    return sum(max(0, min(end, b) - max(start, a)) for a, b in intervals)


def detector_rejections(negative_q, negative_confirmations, event_rates, long_hits, long_total):
    """Only detector/content conditions; raw cache cannot decide original-clean retention."""
    require(long_total > 0 and all(event_rates[k]['total'] > 0
                                  for k in ('short_all', 'short_weak', 'long_all')), 'EMPTY_REQUIRED_DENOMINATOR')
    rejected = []
    if negative_q != 0:
        rejected.append('NEGATIVE_SOURCE_QUALIFICATION_NONZERO')
    if negative_confirmations != 0:
        rejected.append('NEGATIVE_CONFIRMATION_NONZERO')
    for k in ('short_all', 'short_weak', 'long_all'):
        if 100 * event_rates[k]['hits'] < 95 * event_rates[k]['total']:
            rejected.append(k.upper() + '_TIMELY_CONTENT_PROXY_BELOW95')
    if 100 * long_hits < 90 * long_total:
        rejected.append('LONG_ACTIVITY_FRAME_QUALIFICATION_BELOW90')
    return rejected


def validate_meta(meta, recipe, entry, rows, labels, np):
    require(meta['schema_version'] == 1 and meta['scene_id'] == entry['scene_id'] == recipe['id']
            and meta['partition'] == recipe['partition'] == entry['partition'] == 'DEV-CAL'
            and meta['recipe_sha256'] == recipe['recipe_sha256'] == entry['scene_recipe_sha256'],
            'EVENT_RECIPE_IDENTITY')
    body = {k: v for k, v in recipe.items() if k != 'recipe_sha256'}
    require(hashlib.sha256(canonical(body)).hexdigest() == recipe['recipe_sha256'], 'RECIPE_CONTENT_SHA')
    require(meta['subtype'] == recipe['kind'] in ('echo', 'noise', 'long_double', 'short_double', 'near_only')
            and type(meta['weak_double']) is bool and meta['weak_double'] == recipe['weak_double']
            and meta['near_duration_ms'] == recipe['near_duration_ms']
            and meta['raw48_samples'] == RAW_SAMPLES, 'EVENT_SUBGROUP_OR_SOURCE_EXTENT')
    if meta['weak_double']:
        require(meta['subtype'] in ('long_double', 'short_double')
                and -20 <= recipe['near_echo_db'] <= -10, 'WEAK_RECIPE_MEMBERSHIP')
    elif meta['subtype'] in ('long_double', 'short_double'):
        require(-10 <= recipe['near_echo_db'] <= 10, 'NONWEAK_RECIPE_MEMBERSHIP')
    require(type(meta['ADC_clipped']) is bool and type(meta['ADC_clipped_samples']) is int
            and meta['ADC_clipped'] == (meta['ADC_clipped_samples'] > 0), 'ADC_METADATA')
    intervals = meta['active_intervals_raw48']
    last, mask = 0, np.zeros(RAW_SAMPLES, dtype=np.bool_)
    for item in intervals:
        require(isinstance(item, list) and len(item) == 2
                and all(type(v) is int for v in item), 'ACTIVITY_INTERVAL_TYPE')
        a, b = item
        require(0 <= a < b <= RAW_SAMPLES and a >= last, 'ACTIVITY_INTERVAL_ORDER')
        mask[a:b] = True
        last = b
    count = sum(b - a for a, b in intervals)
    require(hashlib.sha256(mask.tobytes()).hexdigest() == meta['activity_sha256'], 'ACTIVITY_IDENTITY')
    require(len(meta['audio_sha256']) == 64 and all(c in '0123456789abcdef' for c in meta['audio_sha256']),
            'AUDIO_PAYLOAD_IDENTITY_REFERENCE')
    event = meta['near_event']
    if meta['subtype'] in ('echo', 'noise'):
        require(event is None and not intervals and not meta['weak_double'], 'NEGATIVE_SOURCE_HAS_ACTIVITY')
    else:
        require(intervals and event is not None and event['first_active_raw48'] == intervals[0][0]
                and event['last_active_raw48'] == intervals[-1][1] - 1
                and event['end_exclusive_raw48'] == intervals[-1][1]
                and event['active_samples'] == count
                and event['definition'] == 'Independent clean source 5ms RMS proxy before gain; identity near path',
                'NEAR_EVENT_SOURCE_BINDING')
    valid = rows[:, 10] == 1
    expected = np.asarray([overlap(intervals, int(r[2]), int(r[3])) / HOP for r in rows[valid]], '<f4')
    require(np.array_equal(expected.view('<u4'), labels.view('<u4')), 'SOURCE_LABEL_BITS')
    represented = sum(overlap(intervals, int(r[2]), int(r[3])) for r in rows[valid])
    require(meta['producer_valid_score_segments'] == entry['producer_segments']
            and meta['invalid_hops'] == np.flatnonzero(~valid).tolist() and meta['source_activity_samples'] == count
            and meta['represented_activity_samples'] == represented
            and meta['unsupported_activity_samples'] == count - represented,
            'SOURCE_COVERAGE_METADATA')
    rounding = float(np.sum(labels, dtype=np.float64) * HOP) - represented
    require(rounding == meta['fractional_label_rounding_samples'], 'FRACTIONAL_LABEL_ACCOUNTING')
    require(meta['subtype'] != 'short_double' or meta['near_duration_ms'] in (320, 400, 480),
            'SHORT_DENOMINATOR_NOMINAL_DURATION')
    return intervals


def source_slots(rows, raw_end, intervals, reset_events=()):
    """Every native target is retained; EOF truth slots never acquire a score."""
    reset_rows = {}
    for event in reset_events:
        require(len(event) == 8 and event[7] == 2, 'RESET_LEDGER_RETURN')
        matches = [i for i,r in enumerate(rows) if int(r[0]) == event[0] and int(r[1]) == event[1]]
        require(len(matches) == 1 and matches[0] not in reset_rows, 'RESET_LEDGER_ROW')
        i = matches[0]
        require([int(rows[i][j]) for j in (2,3,9)] == event[2:5]
                and int(rows[i][10]) == 0 and event[5] == event[0]+1
                and event[6] == event[3]-48, 'RESET_LEDGER_BINDING')
        reset_rows[i] = event
    slots = []
    for ordinal, row in enumerate(rows):
        r = [int(v) for v in row]
        slots.append({'slot_id': ordinal, 'source_row_id': ordinal, 'epoch': r[0], 'hop': r[1],
                      'target_start': r[2], 'target_end': r[3], 'FIR_start': r[4], 'FIR_end': r[5],
                      'Feature_start': r[6], 'Feature_end': r[7], 'epoch_start': r[8],
                      'input_frontier': r[9], 'valid': r[10] == 1, 'score_index': r[11],
                      'reason': ('VALID' if r[10] == 1 else 'SPX_RESET_CURRENT_REJECTED' if ordinal in reset_rows
                                 else 'RESET_FEATURE_SUPPORT_UNAVAILABLE' if r[0] != 0
                                 else 'STARTUP_FEATURE_SUPPORT_UNAVAILABLE')})
    require(slots and slots[0]['target_start'] >= slots[0]['epoch_start'], 'SOURCE_LATTICE_ORIGIN')
    start = slots[-1]['target_end']
    terminal = reset_rows.get(len(rows)-1)
    eof_epoch = terminal[5] if terminal else slots[-1]['epoch']
    eof_start = terminal[6] if terminal else slots[-1]['epoch_start']
    eof_hop = 0 if terminal else slots[-1]['hop']+1
    while start < raw_end:
        slots.append({'slot_id': len(slots), 'source_row_id': None, 'epoch': eof_epoch,
                      'hop': eof_hop, 'target_start': start, 'target_end': start + HOP,
                      'FIR_start': None, 'FIR_end': None, 'Feature_start': None, 'Feature_end': None,
                      'epoch_start': eof_start, 'input_frontier': None,
                      'valid': False, 'score_index': -1, 'reason': 'NOT_READY_AT_EOF',
                      'identity_status': 'PROSPECTIVE_NOT_EXECUTED'})
        start += HOP; eof_hop += 1
    for i, slot in enumerate(slots):
        require(slot['target_end'] - slot['target_start'] == HOP, 'TARGET_LENGTH')
        if i:
            prev = slots[i - 1]
            require(slot['target_start'] == prev['target_end'], 'SOURCE_TARGET_GAP')
            if slot['epoch'] != prev['epoch']:
                event = reset_rows.get(prev['source_row_id'])
                require(event is not None and [slot['epoch'],slot['epoch_start'],slot['hop']]
                        == [event[5],event[6],0], 'SOURCE_UNEXPLAINED_EPOCH')
        if slot['valid']:
            require(slot['Feature_start'] >= slot['epoch_start']
                    and max(slot['FIR_end'], slot['Feature_end'], slot['target_end']) <= slot['input_frontier']
                    and slot['input_frontier'] <= raw_end, 'SOURCE_CAUSAL_FRONTIER')
        slot['activity_samples'] = overlap(intervals, slot['target_start'], min(raw_end, slot['target_end']))
        slot['actual_target_end'] = min(raw_end, slot['target_end'])
    return slots


def audit_scores(np, probabilities, slots, event, tau_values):
    """One fixed score matrix; vectorized state is equivalent to q3/20 scalar steps."""
    require(probabilities.dtype == np.dtype('<f4') and np.isfinite(probabilities).all()
            and ((probabilities >= 0) & (probabilities <= 1)).all(), 'PROBABILITY_CONTRACT')
    t, n = len(tau_values), len(slots)
    q = np.zeros((t, n), dtype=np.bool_)
    for i, slot in enumerate(slots):
        if slot['valid']:
            require(0 <= slot['score_index'] < len(probabilities), 'SCORE_SOURCE_MAPPING')
            q[:, i] = probabilities[slot['score_index']] >= tau_values
    forward, pre_roll = np.zeros_like(q), np.zeros_like(q)
    pending = np.zeros(t, dtype=np.int16)
    opened, remaining = np.zeros(t, np.bool_), np.zeros(t, np.int16)
    confirmations = []
    event_rows = np.full((t, len(EVENT_COLUMNS)), -1, dtype=np.int64)
    event_rows[:, 3:] = 0
    attributed_count = np.zeros(t, np.int64)
    activity = np.asarray([s['activity_samples'] for s in slots], np.int64)
    for i, slot in enumerate(slots):
        if i and (slots[i - 1]['target_end'] != slot['target_start']
                  or slots[i - 1]['epoch'] != slot['epoch']):
            pending[:] = 0
            opened[:] = False
            remaining[:] = 0
        if not slot['valid']:
            pending[:] = 0
            opened[:] = False
            remaining[:] = 0
            continue
        qualifies = q[:, i]
        pending = np.where(qualifies, np.minimum(pending + 1, CONFIRM), 0).astype(np.int16)
        trigger = (~opened) & qualifies & (pending == CONFIRM)
        if trigger.any():
            witness = slots[i - 2:i + 1]
            require(i >= 2 and len(witness) == CONFIRM and all(s['valid'] for s in witness)
                    and all(witness[j]['target_end'] == witness[j + 1]['target_start']
                            and witness[j]['epoch'] == witness[j + 1]['epoch'] for j in (0, 1)),
                    'CONFIRMATION_WITNESS_CAUSALITY')
            frontier = max(s['input_frontier'] for s in witness)
            oldest = max(slot['epoch_start'], frontier - PRE_ROLL * HOP)
            attributable = event is not None and bool(activity[i - 2:i + 1].sum() > 0)
            delay = frontier - event['first_active_raw48'] if attributable else -1
            onset_retained = attributable and oldest <= event['first_active_raw48'] < frontier
            ids = np.flatnonzero(trigger)
            rows = np.empty((len(ids), len(CONFIRM_COLUMNS)), dtype=np.int64)
            rows[:] = (0, i, i - 2, i - 1, i, frontier, int(attributable), delay, oldest, frontier)
            rows[:, 0] = ids + 1
            confirmations.append(rows)
            if attributable:
                attributed_count[ids] += 1
                first = ids[event_rows[ids, 0] == -1]
                event_rows[first] = (i, frontier, delay, int(0 <= delay <= TIMELY_SAMPLES), int(onset_retained))
            # Original target IDs, clipped raw sample bounds remain in the confirmation ledger.
            retained = [j for j, s in enumerate(slots[:i + 1]) if s['valid'] and s['epoch'] == slot['epoch']
                        and s['target_start'] < frontier and s['target_end'] > oldest]
            if retained:
                pre_roll[np.ix_(ids, retained)] = True
        live = opened | trigger
        forward[:, i] = live & (qualifies | (remaining > 0))
        remaining = np.where(live & qualifies, HANGOVER,
                             np.where(opened & ~qualifies, np.maximum(remaining - 1, 0), remaining)).astype(np.int16)
        opened = live & (qualifies | (remaining > 0))
    conf = (np.concatenate(confirmations) if confirmations
            else np.empty((0, len(CONFIRM_COLUMNS)), np.int64))
    confirm_counts = np.bincount(conf[:, 0], minlength=t + 1)[1:].astype(np.int64)
    metrics = np.column_stack((q.sum(axis=1), confirm_counts, forward.sum(axis=1), pre_roll.sum(axis=1),
                               (forward | pre_roll).sum(axis=1), q[:, activity == 0].sum(axis=1),
                               confirm_counts - attributed_count,
                               q[:, np.asarray([s['target_start'] >= event['end_exclusive_raw48']
                                                if event else False for s in slots])].sum(axis=1),
                               q[:, activity > 0].sum(axis=1))).astype(np.int64)
    return {'q': q, 'current_forward': forward, 'pre_roll_intersection': pre_roll,
            'confirmations': conf, 'event': event_rows, 'metrics': metrics}


def scalar_audit(probabilities, slots, event, tau):
    """Independent tiny-fixture reference; no NumPy, model or source audio."""
    q, forward, pre, confirmations = [], [], set(), []
    pending, opened, remaining = 0, False, 0
    earliest = [-1, -1, -1, 0, 0]
    for i, slot in enumerate(slots):
        if i and (slots[i - 1]['target_end'] != slot['target_start']
                  or slots[i - 1]['epoch'] != slot['epoch']):
            pending, opened, remaining = 0, False, 0
        qualifies = slot['valid'] and probabilities[slot['score_index']] >= tau
        q.append(bool(qualifies))
        if not slot['valid']:
            pending, opened, remaining = 0, False, 0
            forward.append(False)
            continue
        pending = min(pending + 1, CONFIRM) if qualifies else 0
        trigger = not opened and qualifies and pending == CONFIRM
        if trigger:
            frontier = max(s['input_frontier'] for s in slots[i - 2:i + 1])
            oldest = max(slot['epoch_start'], frontier - PRE_ROLL * HOP)
            attributable = event is not None and any(s['activity_samples'] > 0 for s in slots[i - 2:i + 1])
            delay = frontier - event['first_active_raw48'] if attributable else -1
            confirmations.append([1, i, i - 2, i - 1, i, frontier, int(attributable), delay, oldest, frontier])
            for j, saved in enumerate(slots[:i + 1]):
                if (saved['valid'] and saved['epoch'] == slot['epoch']
                        and saved['target_start'] < frontier and saved['target_end'] > oldest):
                    pre.add(j)
            if attributable and earliest[0] == -1:
                earliest = [i, frontier, delay, int(0 <= delay <= TIMELY_SAMPLES),
                            int(oldest <= event['first_active_raw48'] < frontier)]
        live = opened or trigger
        forward.append(bool(live and (qualifies or remaining > 0)))
        if live and qualifies:
            remaining = HANGOVER
        elif opened and not qualifies:
            remaining = max(remaining - 1, 0)
        opened = live and (qualifies or remaining > 0)
    return q, forward, [i in pre for i in range(len(slots))], confirmations, earliest


def fixture_check(destination):

            mutate_slots(slots)
        event = ({'first_active_raw48': intervals[0][0], 'end_exclusive_raw48': intervals[-1][1]}
                 if intervals else None)
        probs = np.asarray(values, '<f4')
        taus = np.asarray([i / 1000 for i in range(1, 1000)], '<f4')
        result = audit_scores(np, probs, slots, event, taus)

            actual_conf = result['confirmations'][result['confirmations'][:, 0] == j + 1].copy()

                    and result['current_forward'][j].tolist() == truth[1]

             require(r['confirmations'][r['confirmations'][:,0] == j+1,1].tolist() == [2,8]

def append_group(groups, name, tau_count, np):
    if name not in groups:
        groups[name] = {'events': 0, 'timely_hits': np.zeros(tau_count, np.int64),
                        'raw_onset_misses_when_timely': np.zeros(tau_count, np.int64)}
    return groups[name]


def scene_groups(meta, recipe):
    names = [meta['subtype']]
    if meta['subtype'] == 'short_double':
        names.extend(('short_all', 'short_' + str(meta['near_duration_ms']) + 'ms'))
        if meta['weak_double']:
            names.append('short_weak')
    if meta['subtype'] == 'long_double':
        names.append('long_all')
        if meta['weak_double']:
            names.append('long_weak')
    if meta['subtype'] in ('long_double', 'short_double') and meta['weak_double']:
        names.append('double_weak')
    if meta['near_event']:
        names.extend(('ADC_clipped' if meta['ADC_clipped'] else 'ADC_unclipped',
                      'startup_near' if meta['near_event']['first_active_raw48'] == 0 else 'nonstartup_near'))
        if recipe['path_change_raw48'] is not None:
            names.append('path_change_near')
    return names

            row = {'tau_index': j + 1, 'tau_decimal': (j + 1) / 1000,
                   'tau_Float32_bits': int(tau_values.view('<u4')[j]),
                   'negative_q': int(negatives_q[j]), 'negative_confirmations': int(negatives_confirm[j]),
                   'long_activity_frames': {'hits': int(long_q[j]), 'total': long_total},
                   'groups': {k: {'events': v['events'], 'timely_content_proxy_hits': int(v['timely_hits'][j]),
                                 'raw_onset_misses_when_timely': int(v['raw_onset_misses_when_timely'][j])}
                              for k, v in groups.items()},
                   'near_scene_extra_errors': {'zero_activity_q': int(near_zero_q[j]),
                        'unmatched_confirmations': int(unmatched[j]), 'post_event_q': int(tail_q[j])},
                   'raw_onset_misses_when_timely': int(all_timely_retention_misses[j]),
                   'raw_retention_proxy_status': ('ONSET_PROXY_MISS' if all_timely_retention_misses[j]
                       else 'NO_RAW_ONSET_PROXY_MISS_OBSERVED'), 'raw_retention_proxy_gate_applied': False,
                   'detector_content_proxy_candidate': not rejected,
                   'detector_content_proxy_rejected_conditions': rejected,
                   'complete_original_gates': 'UNVERIFIED', 'final_frozen_tau': None}
            table.append(row)
            if not rejected:
                candidates.append(j + 1)
        for p, expected in frozen.items():
            require(sha(p) == expected, 'FROZEN_INPUT_CHANGED_AT_COMPLETION_' + p)
        require(prediction_count == grid_count == 500 and module.peak_rss() <= RSS_LIMIT,
                'INCOMPLETE_CAL_OR_RESOURCE_LIMIT')
        publish_json(out / 'cal-threshold-candidates.json', {'schema_version': 1, 'thresholds': table,
                     'detector_content_proxy_candidate_tau_indices': candidates, 'final_frozen_tau': None,
                     'raw_retention_proxy_gate_applied': False})
        publish_json(out / 'output-identities.json', {'schema_version': 1, 'scenes': asset_results})
        status = 'PROVISIONAL_CONTENT_PROXY_ONLY' if candidates else 'NO_CONTENT_PROXY_THRESHOLD'
        result = {'schema_version': 1, 'status': status, 'scene_predictions': prediction_count,
                  'scenes_grid_complete': grid_count, 'thresholds': 999, 'groups': group_counts,
                  'min_content_proxy_tau': min(candidates) / 1000 if candidates else None,
                  'min_content_proxy_tau_Float32_bits': int(tau_values.view('<u4')[min(candidates)-1]) if candidates else None,
                  'minimum_selection_basis': 'DETECTOR_CONTENT_GATES_ONLY_RAW_RETENTION_DIAGNOSTIC',
                  'final_frozen_tau': None, 'threshold_frozen': False, 'actual_ready': 'UNVERIFIED',
                  'original_clean_PCM': 'UNVERIFIED', 'trained_native_performance': 'UNVERIFIED',
                  'native_sigmoid_qualification_transfer': 'UNVERIFIED', 'DEV_LOCK': 'NOT_ADMITTED',
                  'PUBLIC_TEST': 'NOT_ADMITTED', 'elapsed_seconds': time.monotonic() - started,
                  'peak_RSS_bytes': module.peak_rss(), 'pre_run_sha256': sha(out / 'pre-run-identity.json'),
                  'cal_threshold_candidates_sha256': sha(out / 'cal-threshold-candidates.json'),
                  'output_identities_sha256': sha(out / 'output-identities.json'),
                  'failure': None, 'Stage7': 'PASS', 'Test3': 'BLOCKED', 'NO_PRODUCT_PATCH': True}
        publish_json(out / 'evaluation-result.json', result)
        print(json.dumps({'status': status, 'min_content_proxy_tau': result['min_content_proxy_tau'],
                          'final_frozen_tau': None, 'Test3': 'BLOCKED'}), flush=True)
        return 0
    except Exception as error:
        failure = str(error)
        publish_json(out / 'evaluation-rejected.json', {'schema_version': 1, 'status': 'INSUFFICIENT_EVIDENCE',
                     'failure': failure, 'scene_predictions': prediction_count, 'scenes_grid_complete': grid_count,
                     'final_frozen_tau': None, 'elapsed_seconds': time.monotonic() - started,
                     'actual_ready': 'UNVERIFIED', 'original_clean_PCM': 'UNVERIFIED',
                     'DEV_LOCK': 'NOT_ADMITTED', 'Test3': 'BLOCKED', 'NO_PRODUCT_PATCH': True})
        print(json.dumps({'status': 'INSUFFICIENT_EVIDENCE', 'failure': failure}), flush=True)
        return 1


def main():
    parser = argparse.ArgumentParser()
    sub = parser.add_subparsers(dest='command', required=True)
    fixture = sub.add_parser('fixture-check')
    fixture.add_argument('new_output')
    cal = sub.add_parser('calibrate')
    for name in ('feature_index', 'checkpoint', 'model_manifest', 'train_module', 'new_output'):
        cal.add_argument(name)
    cal.add_argument('--authorization', default=str(AUTHORIZATION))
    cal.add_argument('--approved-spx-nh1-dev-cal', action='store_true', required=True)
    lock = sub.add_parser('dev-lock')
    lock.add_argument('freeze_manifest')
    args = parser.parse_args()
    if args.command == 'fixture-check':
        return fixture_check(args.new_output)
    if args.command == 'dev-lock':
        require(False, 'DEV_LOCK_NOT_ADMITTED_COMPLETE_ORIGINAL_GATES_AND_FINAL_TAU_REQUIRED')
    return calibration(args.feature_index, args.checkpoint, args.model_manifest, args.train_module,
                       args.new_output, args.authorization)

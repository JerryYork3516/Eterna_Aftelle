"""Inherited census and sampled aggregate RSS; no execution at import."""
import os
from pathlib import Path
import subprocess
from rbt_common import D

def storage(public, reserve):
    total=0; files=0
    for directory,dirs,names in os.walk(public,followlinks=False):
        if any((Path(directory)/p).is_symlink() for p in dirs+names):
            raise RuntimeError('SOR_PUBLIC_SYMLINK')
        for name in names:
            total+=(Path(directory)/name).stat().st_size; files+=1
    local=os.statvfs(D); external=os.statvfs(public)
    available=local.f_bavail*local.f_frsize
    if total>60*2**30 or available<reserve or external.f_bavail*external.f_frsize<=0:
        raise RuntimeError('NOT_RUN_RESOURCE')
    return dict(public_logical_bytes=total,public_files=files,public_cap=60*2**30,
                new_public_bytes=0,local_available_bytes=available,local_required_reserve=reserve,
                public_available_bytes=external.f_bavail*external.f_frsize)

def process_usage(root, group=None):
    rows=[]
    result=subprocess.run(['ps','-axo','pid=,ppid=,pgid=,rss='],capture_output=True,text=True,check=True,timeout=1)
    for line in result.stdout.splitlines():
        p,parent,pgid,rss=map(int,line.split()); rows.append((p,parent,pgid,rss*1024))
    if root not in {p for p,_,_,_ in rows}:
        raise RuntimeError('SOR_MONITOR_ROOT_MISSING')
    children={root}
    while True:
        updated=children|{p for p,parent,_,_ in rows if parent in children}
        if updated==children: break
        children=updated
    members={p for p,_,g,_ in rows if group is not None and g==group}
    return sum(rss for p,_,_,rss in rows if p in children|members),sorted(members)
